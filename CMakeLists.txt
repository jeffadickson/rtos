cmake_minimum_required(VERSION 3.22)
include(cmake/platform.cmake)
project(riscue_rtos VERSION 0.2.0 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(cmake/configuration.cmake)
include(cmake/host-memory.cmake)
include(cmake/host-tools.cmake)
nexrt_add_host_tool(rtos_tools Rtos.Tools tools/Rtos.Tools/Rtos.Tools.csproj)
nexrt_add_host_tool(evidence_tool NexRT.Evidence tools/NexRT.Evidence/NexRT.Evidence.csproj)
nexrt_add_host_tool(handoff_tool NexRT.Handoff tools/NexRT.Handoff/NexRT.Handoff.csproj)
nexrt_add_host_tool(network_test_tool NexRT.NetTest tools/NexRT.NetTest/NexRT.NetTest.csproj)
add_library(rtos_options INTERFACE)
target_include_directories(rtos_options INTERFACE include src)
add_library(nexrt::options ALIAS rtos_options)
target_compile_features(rtos_options INTERFACE cxx_std_20)
foreach(key MAX_THREADS TICK_HZ IDLE_STACK_BYTES MTIME_ADDRESS MTIMECMP_ADDRESS IPC_MAX_MESSAGE_BYTES MAX_HELD_MUTEXES)
  target_compile_definitions(rtos_options INTERFACE RTOS_${key}=${RTOS_${key}})
endforeach()
foreach(key ROUND_ROBIN METRICS STACK_GUARD STACK_WATERMARK SYNC LOGGING DMA ZAAMO TASK_TRACE)
  target_compile_definitions(rtos_options INTERFACE RTOS_${key}=$<BOOL:${RTOS_${key}}>)
endforeach()
target_compile_definitions(rtos_options INTERFACE RTOS_TASK_TRACE_CAPACITY=${RTOS_TASK_TRACE_CAPACITY})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(rtos_options INTERFACE -Wall -Wextra -Werror)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(rtos_options INTERFACE $<$<CONFIG:Release>:-O3;-ftree-loop-vectorize;-ftree-slp-vectorize>)
  endif()
endif()
include(cmake/modules.cmake)
include(cmake/static-services.cmake)
if(RTOS_BUILD_FIRMWARE)
  include(cmake/firmware.cmake)
else()
  enable_testing()
  include(cmake/host-tests.cmake)
  option(RTOS_INTERPRETER "Build the C++20 RISC-V instruction model" OFF)
  if(RTOS_INTERPRETER)
    add_subdirectory(sim/interpreter)
  endif()
endif()
find_program(SPHINX_EXECUTABLE NAMES sphinx-build
  HINTS ${CMAKE_SOURCE_DIR}/build/docs-venv/bin ${CMAKE_SOURCE_DIR}/build/docs-venv/Scripts)
if(SPHINX_EXECUTABLE)
  add_custom_target(docs
    COMMAND ${SPHINX_EXECUTABLE} -n -W --keep-going -b html
      ${CMAKE_SOURCE_DIR}/docs ${CMAKE_BINARY_DIR}/docs/html VERBATIM)
else()
  add_custom_target(docs COMMAND ${CMAKE_COMMAND} -E echo
    "Install docs/requirements.txt and reconfigure to locate sphinx-build."
    COMMAND ${CMAKE_COMMAND} -E false VERBATIM)
endif()

include(cmake/requirements.cmake)

include(cmake/static-analysis.cmake)

include(cmake/configure-tool.cmake)

if(RTOS_NATIVE_TOOLS)
  add_custom_target(native_tools DEPENDS rtos_tools evidence_tool handoff_tool requirements_tool configure_tool)
endif()

nexrt_add_host_tool(workbench_tool NexRT.Workbench tools/NexRT.Workbench/NexRT.Workbench.csproj)
nexrt_add_host_tool(services_tool NexRT.Services tools/NexRT.Services/NexRT.Services.csproj)
if(NOT RTOS_BUILD_FIRMWARE)
  add_test(NAME optional_services_configuration COMMAND ${services_tool_COMMAND} configuration ${CMAKE_BINARY_DIR})
endif()

if(RTOS_NATIVE_TOOLS)
  add_dependencies(native_tools workbench_tool services_tool network_test_tool)
endif()

option(RTOS_SIMULATION "Build native C# RISCue/SharpTestBench simulation tooling" OFF)
if(RTOS_SIMULATION)
  nexrt_add_host_tool(simulation_tool NexRT.Sim tools/NexRT.Sim/NexRT.Sim.csproj)
  if(RTOS_NATIVE_TOOLS)
    add_dependencies(native_tools simulation_tool)
  endif()
  set(RISCUE_SOURCE_DIR "${CMAKE_SOURCE_DIR}/../RISCue" CACHE PATH "Existing RISCue checkout for simulation")
  set(SHARPTESTBENCH_SOURCE_DIR "${CMAKE_SOURCE_DIR}/../SharpTestBench" CACHE PATH "Existing SharpTestBench checkout")
  set(RTOS_SIM_ELF ${CMAKE_SOURCE_DIR}/build/uart/demo.elf CACHE FILEPATH "Firmware for UART simulation")
  add_custom_target(sharp_demo
    COMMAND ${simulation_tool_COMMAND} ${CMAKE_SOURCE_DIR} ${RISCUE_SOURCE_DIR} ${SHARPTESTBENCH_SOURCE_DIR} ${RTOS_SIM_ELF}
    DEPENDS simulation_tool USES_TERMINAL VERBATIM)
  set(RTOS_SOAK_ELF ${CMAKE_SOURCE_DIR}/build/overnight-stress/demo.elf CACHE FILEPATH "Firmware for the randomized overnight simulation")
  foreach(mode prepare-soak smoke soak)
    string(REPLACE "-" "_" target_mode "${mode}")
    add_custom_target(sharp_${target_mode}
      COMMAND ${simulation_tool_COMMAND} ${CMAKE_SOURCE_DIR} ${RISCUE_SOURCE_DIR} ${SHARPTESTBENCH_SOURCE_DIR} ${RTOS_SOAK_ELF} --${mode}
      DEPENDS simulation_tool USES_TERMINAL VERBATIM)
  endforeach()
  set(RISCUE_ZAAMO_SOURCE_DIR "${CMAKE_SOURCE_DIR}/../RISCue-Zaamo" CACHE PATH "Separate source snapshot with optional Zaamo support")
  foreach(variant baseline zaamo)
    add_custom_target(sharp_atomics_${variant}
      COMMAND ${simulation_tool_COMMAND} ${CMAKE_SOURCE_DIR} ${RISCUE_ZAAMO_SOURCE_DIR} ${SHARPTESTBENCH_SOURCE_DIR}
        ${CMAKE_SOURCE_DIR}/build/atomic-${variant}/demo.elf --atomics-${variant}
      DEPENDS simulation_tool USES_TERMINAL VERBATIM)
  endforeach()
endif()
