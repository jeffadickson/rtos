cmake_minimum_required(VERSION 3.22)
include(cmake/platform.cmake)
project(riscue_rtos VERSION 0.2.0 LANGUAGES CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(cmake/configuration.cmake)
include(cmake/host-tools.cmake)
nexrt_add_host_tool(rtos_tools Rtos.Tools tools/Rtos.Tools/Rtos.Tools.csproj)
add_library(rtos_options INTERFACE)
target_include_directories(rtos_options INTERFACE include src)
add_library(nexrt::options ALIAS rtos_options)
foreach(key MAX_THREADS TICK_HZ IDLE_STACK_BYTES MTIME_ADDRESS MTIMECMP_ADDRESS)
  target_compile_definitions(rtos_options INTERFACE RTOS_${key}=${RTOS_${key}})
endforeach()
foreach(key ROUND_ROBIN METRICS STACK_GUARD SYNC)
  target_compile_definitions(rtos_options INTERFACE RTOS_${key}=$<BOOL:${RTOS_${key}}>)
endforeach()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(rtos_options INTERFACE -Wall -Wextra -Werror)
endif()
include(cmake/modules.cmake)
if(RTOS_BUILD_FIRMWARE)
  if(NOT CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Firmware requires cmake/riscue-toolchain.cmake")
  endif()
  add_library(rtos STATIC src/kernel.cpp src/memory.cpp ports/riscue/context.S)
  target_link_libraries(rtos PUBLIC rtos_options)
  if(RTOS_SYNC)
    target_sources(rtos PRIVATE src/sync.cpp)
  endif()
  target_include_directories(rtos PRIVATE ports/riscue)
  add_library(nexrt::kernel ALIAS rtos)
  target_compile_options(rtos PRIVATE -Os -fstack-usage -ffunction-sections -fdata-sections)
  set(RTOS_APPLICATION examples/blink.cpp CACHE FILEPATH "Application source")
  add_executable(demo ${RTOS_APPLICATION} ports/riscue/start.S)
  target_compile_options(demo PRIVATE -fstack-usage)
  target_link_libraries(demo PRIVATE rtos gcc)
  set_target_properties(demo PROPERTIES SUFFIX .elf)
  target_link_options(demo PRIVATE -nostdlib -T${CMAKE_SOURCE_DIR}/ports/riscue/link.ld
    -Wl,--gc-sections,--no-relax,-Map,${CMAKE_BINARY_DIR}/demo.map)
  set_property(TARGET demo APPEND PROPERTY LINK_DEPENDS ${CMAKE_SOURCE_DIR}/ports/riscue/link.ld)
  set_property(TARGET demo APPEND PROPERTY LINK_DEPENDS ${rtos_tools_OUTPUT})
  add_dependencies(demo rtos_tools)
  add_custom_command(TARGET demo POST_BUILD
    COMMAND ${rtos_tools_COMMAND} audit
      --nm ${CMAKE_NM} --elf $<TARGET_FILE:demo>
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:demo> VERBATIM)
  add_custom_target(static_contract_check
    COMMAND ${rtos_tools_COMMAND} static-contract
      --cxx ${CMAKE_CXX_COMPILER} --nm ${CMAKE_NM}
      --source-root ${CMAKE_SOURCE_DIR} --build-dir ${CMAKE_BINARY_DIR}/static-contract
    DEPENDS rtos_tools VERBATIM)
else()
  enable_testing()
  if(DOTNET_EXECUTABLE)
    add_test(NAME tooling COMMAND ${rtos_tools_COMMAND} self-test)
  endif()
  add_executable(memory_test tests/memory.cpp src/memory.cpp)
  target_link_libraries(memory_test PRIVATE rtos_options)
  if(MSVC)
    target_compile_options(memory_test PRIVATE /UNDEBUG /Oi-)
  else()
    target_compile_options(memory_test PRIVATE -UNDEBUG -fno-builtin)
  endif()
  add_test(NAME memory COMMAND memory_test)
  add_executable(scheduler_test tests/scheduler.cpp)
  target_link_libraries(scheduler_test PRIVATE rtos_options)
  # Assertions must execute even in release builds.
  if(MSVC)
    target_compile_options(scheduler_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(scheduler_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME scheduler COMMAND scheduler_test)
  if(RTOS_ETHERNET)
    add_executable(ethernet_test tests/ethernet.cpp)
    target_link_libraries(ethernet_test PRIVATE rtos_ethernet)
    if(MSVC)
      target_compile_options(ethernet_test PRIVATE /UNDEBUG)
    else()
      target_compile_options(ethernet_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ethernet COMMAND ethernet_test)
  endif()
endif()
find_program(SPHINX_EXECUTABLE NAMES sphinx-build
  HINTS ${CMAKE_SOURCE_DIR}/build/docs-venv/bin ${CMAKE_SOURCE_DIR}/build/docs-venv/Scripts)
if(SPHINX_EXECUTABLE)
  add_custom_target(docs
    COMMAND ${SPHINX_EXECUTABLE} -n -W --keep-going -b html
      ${CMAKE_SOURCE_DIR}/docs ${CMAKE_BINARY_DIR}/docs/html VERBATIM)
else()
  add_custom_target(docs COMMAND ${CMAKE_COMMAND} -E echo
    "Install docs/requirements.txt and reconfigure to locate sphinx-build."
    COMMAND ${CMAKE_COMMAND} -E false VERBATIM)
endif()

include(cmake/requirements.cmake)

include(cmake/static-analysis.cmake)

include(cmake/configure-tool.cmake)

if(RTOS_NATIVE_TOOLS)
  add_custom_target(native_tools DEPENDS rtos_tools requirements_tool configure_tool)
endif()

nexrt_add_host_tool(workbench_tool NexRT.Workbench tools/NexRT.Workbench/NexRT.Workbench.csproj)

if(RTOS_NATIVE_TOOLS)
  add_dependencies(native_tools workbench_tool)
endif()

if(NOT RTOS_BUILD_FIRMWARE)
  add_executable(static_test tests/static.cpp)
  target_link_libraries(static_test PRIVATE rtos_options)
  if(NOT MSVC)
    target_compile_options(static_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME static_storage COMMAND static_test)
endif()

if(RTOS_WEB)
  if(RTOS_BUILD_FIRMWARE)
    add_executable(web_link_probe examples/web_probe.cpp ports/riscue/start.S)
    target_include_directories(web_link_probe PRIVATE generated)
    target_compile_options(web_link_probe PRIVATE -Os)
    target_link_libraries(web_link_probe PRIVATE rtos_web rtos gcc)
    target_link_options(web_link_probe PRIVATE -nostdlib -T${CMAKE_SOURCE_DIR}/ports/riscue/link.ld -Wl,--gc-sections,--no-relax)
    set_target_properties(web_link_probe PROPERTIES SUFFIX .elf)
    add_dependencies(web_link_probe rtos_tools)
    add_custom_command(TARGET web_link_probe POST_BUILD COMMAND ${rtos_tools_COMMAND} audit --nm ${CMAKE_NM} --elf $<TARGET_FILE:web_link_probe> VERBATIM)
  else()
    add_executable(web_test tests/web.cpp)
    target_link_libraries(web_test PRIVATE rtos_web)
    if(NOT MSVC)
      target_compile_options(web_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME web COMMAND web_test)
  endif()
endif()

if(NOT RTOS_BUILD_FIRMWARE)
  add_executable(sync_test tests/sync.cpp src/sync.cpp)
  target_include_directories(sync_test PRIVATE include src tests)
  target_compile_definitions(sync_test PRIVATE RTOS_SYNC=1)
  target_compile_options(sync_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME synchronization COMMAND sync_test)
endif()
