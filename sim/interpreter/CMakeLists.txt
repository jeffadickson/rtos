cmake_minimum_required(VERSION 3.22)
project(NexRT_Interpreter LANGUAGES CXX)
# Standalone build avoids touching any live RTL launcher, model or firmware.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RTOS_SERVICES_ELF "" CACHE FILEPATH "Optional scheduled static-service firmware acceptance image")
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
add_library(nexrt_rv_model machine.cpp elf.cpp)
target_include_directories(nexrt_rv_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(nexrt_rv_model PUBLIC cxx_std_20)
option(NEXRT_INTERPRETER_SANITIZERS "Enable host address/undefined-behavior checks" OFF)
if(NEXRT_INTERPRETER_SANITIZERS AND NOT MSVC)
  target_compile_options(nexrt_rv_model PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nexrt_rv_model PUBLIC -fsanitize=address,undefined)
endif()
if(MSVC)
  target_compile_options(nexrt_rv_model PUBLIC /W4 /WX $<$<CONFIG:Release>:/O2>)
else()
  target_compile_options(nexrt_rv_model PUBLIC -Wall -Wextra -Werror $<$<CONFIG:Release>:-O3;-ftree-loop-vectorize;-ftree-slp-vectorize>)
endif()
add_executable(nexrt-rv main.cpp)
target_link_libraries(nexrt-rv PRIVATE nexrt_rv_model)
add_executable(nexrt-rv-tests tests.cpp)
target_link_libraries(nexrt-rv-tests PRIVATE nexrt_rv_model)
enable_testing()
add_test(NAME rv_architecture COMMAND nexrt-rv-tests)
if(RTOS_SERVICES_ELF)
  if(NOT EXISTS "${RTOS_SERVICES_ELF}")
    message(FATAL_ERROR "RTOS_SERVICES_ELF does not exist; build services-rv32-sim first")
  endif()
  add_test(NAME rv_static_services COMMAND nexrt-rv --elf ${RTOS_SERVICES_ELF}
    --ram-bytes 262144 --host-seconds 60 --max-instructions 200000000)
  set_tests_properties(rv_static_services PROPERTIES TIMEOUT 75)
  add_test(NAME rv_static_services_cutoff COMMAND nexrt-rv --elf ${RTOS_SERVICES_ELF}
    --ram-bytes 262144 --host-seconds 1 --max-instructions 1)
  set_tests_properties(rv_static_services_cutoff PROPERTIES WILL_FAIL TRUE TIMEOUT 5)
  add_test(NAME rv_static_services_wrong_clock COMMAND nexrt-rv --elf ${RTOS_SERVICES_ELF}
    --ram-bytes 262144 --clock-hz 1 --host-seconds 1 --max-instructions 1)
  set_tests_properties(rv_static_services_wrong_clock PROPERTIES WILL_FAIL TRUE TIMEOUT 5)
endif()
